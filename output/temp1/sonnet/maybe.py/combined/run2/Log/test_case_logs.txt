    """Test that Maybe(None, True) correctly chains through ap, to_lazy, to_validation, filter, get_or_else, to_either, to_try, __eq__, and to_box without errors."""

    validation_from_lazy = lazy_from_applied.to_validation()

    # Unwrap filtered_maybe, falling back to itself if empty

    # Convert the Validation into a Try
    try_from_validation = validation_from_lazy.to_try()

    # Compare filtered_maybe with applied_maybe for equality (result not asserted)
    eq_result = filtered_maybe.__eq__(applied_maybe)

    # Wrap the unwrapped value in a Box
    boxed_unwrapped = unwrapped_or_else.to_box()

    # Apply raw bytes to the Try instance
    try_from_validation.ap(raw_bytes_input)
