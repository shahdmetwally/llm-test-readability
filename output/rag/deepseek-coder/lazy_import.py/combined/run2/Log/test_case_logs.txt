def test_timer_start_and_stop_functionality():

      default='a',
      metavar='SYMBOL',

  )
  parser_object.add_argument(
      '--lazy-option',
      help="the lazy option",
      default='a',
      metavar='SYMBOL',
      action='lazy_choices',
      getter=getter_object,
      cache=False
  )

  parser_object.parse_args([])
  getter_object.assert_not_called()

  getter_object.assert_not_called()

  getter_object.assert_called()

  with pytest.raises(SystemExit):
